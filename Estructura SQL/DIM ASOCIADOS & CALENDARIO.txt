-- ==========================================
-- TABLA MADRE: PERSONAS (dim_Asociados)
-- ==========================================
CREATE TABLE dim_Asociados (
    ID_Asociado VARCHAR(10) PRIMARY KEY,       -- Formato: GC001, GC002
    Apellidos_Nombres VARCHAR(256) NOT NULL,   -- Orden alfabético nativo garantizado en Power BI
    Fecha_Nacimiento DATE NOT NULL,
    Genero VARCHAR(15) NOT NULL CHECK (Genero IN ('Masculino', 'Femenino')),
    Nivel_Educativo VARCHAR(50) NOT NULL CHECK (Nivel_Educativo IN ('Técnico', 'Bachiller', 'Titulado', 'Maestría')),
    Carrera_Profesional VARCHAR(100) NOT NULL, -- Ej: Psicología, Ingeniería de Sistemas
    Fecha_Ingreso DATE NOT NULL,
    Fecha_Cese DATE NULL,                      -- MVP: Asume cero reingresos (No Rehires)
    
    -- Validaciones de Integridad Estricta
    CHECK (Fecha_Cese >= Fecha_Ingreso)
);

-- ==========================================
-- TABLA MADRE: TIEMPO (dim_Calendario)
-- ==========================================
CREATE TABLE dim_Calendario (
    Fecha_Llave DATE PRIMARY KEY,              -- Granularidad diaria: una fila por fecha
    Anio INT NOT NULL,
    Mes_Numero INT NOT NULL,
    Mes_Nombre VARCHAR(20) NOT NULL,
    Periodo_ID INT NOT NULL,                  -- Ej: 202603 (ID matemático para ordenar ejes en Power BI)
    Etiqueta_Periodo VARCHAR(7) NOT NULL,     -- Ej: Mar2026 (Etiqueta visual de despliegue)
    Trimestre VARCHAR(2) NOT NULL,             -- Ej: Q1, Q2 (Para análisis de estacionalidad histórica)
    Etiqueta_Trimestre VARCHAR(7) NOT NULL,    -- Ej: Q1-2026 (Eje visual continuo para Performance)
    Semestre VARCHAR(2) NOT NULL,              -- Ej: H1, H2 (Para estacionalidad macro de presupuesto)
    Etiqueta_Semestre VARCHAR(7) NOT NULL,     -- Ej: H1-2026 (Eje visual continuo para Finanzas)
    Dia_Semana_Numero INT NOT NULL CHECK (Dia_Semana_Numero BETWEEN 1 AND 7), -- Estándar ISO (1=Lunes, 7=Domingo)
    Dia_Semana_Nombre VARCHAR(10) NOT NULL,   -- Ej: Lunes, Martes... para analítica de ausentismo
    Es_Dia_Laborable BOOLEAN NOT NULL         -- TRUE: Día laborable | FALSE: Fin de semana o Feriado
);
